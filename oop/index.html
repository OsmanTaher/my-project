2.2 كتابة أول كلاس

C++ — BankAccount Classbank_account.cpp 
#include <iostream>
#include <string>
using namespace std;

class BankAccount {
private:                          // 🔒 محمي — لا يُرى من الخارج
    string owner;
    double balance;
    int    accountNumber;

public:                           // 🌐 عام — يمكن الوصول له
    // Constructor
    BankAccount(string ownerName, double initialBalance, int accNum) {
        owner         = ownerName;
        balance       = initialBalance;
        accountNumber = accNum;
    }

    // Methods
    void deposit(double amount) {
        if (amount > 0) {
            balance += amount;
            cout << "✅ تم إيداع " << amount << " جنيه\n";
        } else {
            cout << "❌ المبلغ يجب أن يكون موجباً!\n";
        }
    }

    bool withdraw(double amount) {
        if (amount > 0 && amount <= balance) {
            balance -= amount;
            cout << "✅ تم سحب " << amount << " جنيه\n";
            return true;
        }
        cout << "❌ رصيد غير كافٍ!\n";
        return false;
    }

    void printInfo() const {       // const = لا تعدّل البيانات
        cout << "═══════════════════════\n";
        cout << "الحساب: " << accountNumber << "\n";
        cout << "المالك: " << owner         << "\n";
        cout << "الرصيد: " << balance       << " جنيه\n";
        cout << "═══════════════════════\n";
    }

    double getBalance() const { return balance; }
};

int main() {
    // إنشاء كائنات من الكلاس
    BankAccount acc1("أحمد محمد", 5000.0, 1001);
    BankAccount acc2("سارة علي",  3000.0, 1002);

    acc1.printInfo();
    acc1.deposit(1500);
    acc1.withdraw(200);
    acc1.withdraw(99999); // ❌ مرفوض بأمان

    // ❌ هذا خطأ — لا يمكن الوصول للخاصة مباشرة
    // acc1.balance = -999; // Compile Error!
}
✅ ما تعلمناه في هذا الفصل
الكلاس يجمع البيانات والدوال في وحدة واحدة منظمة
private يحمي البيانات من التعديل العشوائي
public يعرض الواجهة الآمنة للمستخدم
const على الدوال تضمن عدم تعديل البيانات

3.2 Getters & Setters

C++ — Getters & Setterstemperature.cpp 
class Temperature {
private:
    double celsius;     // مخزّن بالسيلزيوس دائماً

public:
    Temperature(double c = 0) : celsius(c) {}

    // Setter مع تحقق
    void setCelsius(double c) {
        if (c < -273.15) {
            cout << "❌ خطأ: أقل من الصفر المطلق!\n";
            return;
        }
        celsius = c;
    }

    // Getters لتحويل الوحدات
    double getCelsius()    const { return celsius; }
    double getFahrenheit() const { return celsius * 9.0/5.0 + 32; }
    double getKelvin()     const { return celsius + 273.15; }

    string getStatus() const {
        if (celsius < 0)   return "❄️ تجميد";
        if (celsius < 20)  return "🌤️ بارد";
        if (celsius < 35)  return "☀️ معتدل";
        return "🔥 حار";
    }

    void print() const {
        cout << getCelsius()    << "°C | "
             << getFahrenheit() << "°F | "
             << getKelvin()     << "K — "
             << getStatus()     << "\n";
    }
};

int main() {
    Temperature t1(25);
    t1.print();               // 25°C | 77°F | 298.15K — ☀️ معتدل

    t1.setCelsius(-300);      // ❌ خطأ: أقل من الصفر المطلق!
    t1.setCelsius(38);
    t1.print();               // 38°C | 100.4°F | 311.15K — 🔥 حار
}

الـ Constructor يُولد الكائن، والـ Destructor يُنهيه. فهمهما بعمق هو الفرق بين مبرمج عادي ومبرمج C++ محترف.

4.1 أنواع الـ Constructors

C++ — كل أنواع Constructorsconstructors.cpp 
class Car {
private:
    string brand, model;
    int    year;
    double price;
    int    mileage;

public:
    // 1️⃣ Default Constructor
    Car() : brand("غير معروف"), model("غير معروف"),
             year(2024), price(0), mileage(0) {
        cout << "🚗 Default Constructor\n";
    }

    // 2️⃣ Parameterized Constructor
    Car(string b, string m, int y, double p)
        : brand(b), model(m), year(y), price(p), mileage(0) {
        cout << "🚗 Parameterized Constructor: " << b << " " << m << "\n";
    }

    // 3️⃣ Copy Constructor
    Car(const Car& other)
        : brand(other.brand), model(other.model),
          year(other.year),   price(other.price),
          mileage(0) {        // السيارة المنسوخة بدون عداد!
        cout << "📋 Copy Constructor: نسخة من " << other.brand << "\n";
    }

    // 4️⃣ Destructor
    ~Car() {
        cout << "💀 Destructor: " << brand << " " << model << "\n";
    }

    void drive(int km) {
        mileage += km;
        cout << "🛣️ قطعت " << km << " كم. الإجمالي: " << mileage << "\n";
    }

    void print() const {
        cout << brand << " " << model
             << " (" << year << ") — "
             << price << " جنيه | "
             << mileage << " كم\n";
    }
};

int main() {
    Car c1;                                    // Default
    Car c2("Toyota", "Corolla", 2023, 450000); // Parameterized
    Car c3 = c2;                               // Copy Constructor

    c2.drive(150);
    c3.drive(50);  // نسخة مستقلة بعداد منفصل

    c2.print();   // 150 كم
    c3.print();   // 50 كم — مستقلة!
} // هنا يُنادى Destructor لكل الكائنات

4.2 Initializer List — الطريقة الاحترافية

⚡قاعدة احترافية: استخدم دائماً Initializer List (: بعد اسم الـ Constructor) بدلاً من التعيين داخل الجسم — أسرع وأكثر صحة، خصوصاً مع const وreferences.
C++ — Initializer List vs Assignment 
class Point {
private:
    const int id;      // const — يجب التهيئة بالـ Initializer List
    double    x, y;

public:
    // ✅ الطريقة الصحيحة — Initializer List
    Point(int _id, double _x, double _y)
        : id(_id), x(_x), y(_y) {}

    double distanceTo(const Point& other) const {
        double dx = x - other.x;
        double dy = y - other.y;
        return sqrt(dx*dx + dy*dy);
    }

    void print() const {
        cout << "نقطة #" << id << ": (" << x << ", " << y << ")\n";
    }
};

CHAPTER 05 — OPERATOR & FUNCTION OVERLOADING

Operator & Function Overloading
اجعل كلاسك يتصرف كالأنواع الأساسية. بدل c1.add(c2) — اكتب c1 + c2. هذا هو ما يجعل C++ قوية ومعبّرة.

5.1 Operator Overloading — الأعداد المركبة

C++ — Complex Numberscomplex.cpp 
class Complex {
private:
    double real, imag;

public:
    Complex(double r = 0, double i = 0) : real(r), imag(i) {}

    // ✅ Overload + للجمع
    Complex operator+(const Complex& other) const {
        return Complex(real + other.real, imag + other.imag);
    }

    // ✅ Overload * للضرب
    Complex operator*(const Complex& other) const {
        return Complex(
            real * other.real - imag * other.imag,
            real * other.imag + imag * other.real
        );
    }

    // ✅ Overload == للمقارنة
    bool operator==(const Complex& other) const {
        return (real == other.real) && (imag == other.imag);
    }

    // ✅ Overload += 
    Complex& operator+=(const Complex& other) {
        real += other.real;
        imag += other.imag;
        return *this;
    }

    // ✅ Overload << للطباعة (friend)
    friend ostream& operator<<(ostream& os, const Complex& c) {
        if (c.imag >= 0)
            os << c.real << " + " << c.imag << "i";
        else
            os << c.real << " - " << -c.imag << "i";
        return os;
    }

    double magnitude() const { return sqrt(real*real + imag*imag); }
};

int main() {
    Complex c1(3, 4), c2(1, -2);

    cout << "c1 = "     << c1             << "\n"; // 3 + 4i
    cout << "c1+c2 = "  << (c1 + c2)     << "\n"; // 4 + 2i
    cout << "c1*c2 = "  << (c1 * c2)     << "\n"; // 11 - 2i
    cout << "|c1| = "   << c1.magnitude() << "\n"; // 5
    cout << (c1 == c2 ? "متساويتان" : "مختلفتان") << "\n";
}

C++ — Inheritance Systememployee.cpp 
class Employee {
protected:                         // 🛡️ يُرث للأبناء
    string name, department;
    int    id;
    double baseSalary;

public:
    Employee(string n, int i, double s, string d)
        : name(n), id(i), baseSalary(s), department(d) {}

    virtual double getSalary() const { return baseSalary; }
    virtual void   print()     const {
        cout << "[ID:" << id << "] " << name
             << " | " << department
             << " | " << getSalary() << " جنيه\n";
    }
    virtual ~Employee() {}       // ضروري مع virtual
};

// ────────────────────────────────────────────
class Manager : public Employee {
private:
    double bonus;
    int    teamSize;

public:
    Manager(string n, int i, double s, string d, double b, int t)
        : Employee(n, i, s, d), bonus(b), teamSize(t) {}

    double getSalary() const override {
        return baseSalary + bonus;
    }

    void print() const override {
        Employee::print();              // استدعاء الأب
        cout << "  📋 يدير فريقاً من " << teamSize << " أشخاص\n";
    }
};

class Developer : public Employee {
private:
    string language;
    int    level;   // 1=Junior, 2=Mid, 3=Senior

public:
    Developer(string n, int i, double s, string lang, int lvl)
        : Employee(n, i, s, "Engineering"), language(lang), level(lvl) {}

    double getSalary() const override {
        return baseSalary * (1.0 + level * 0.15);
    }
};

int main() {
    Manager   mgr("أحمد", 1, 15000, "IT", 5000, 8);
    Developer dev("سارة", 2, 12000, "C++", 3);

    mgr.print();  // 20000 جنيه
    dev.print();  // 17400 جنيه (Senior)
}

7.1 Runtime Polymorphism — الجوهر

💡 المفهوم

باستخدام pointer أو reference للكلاس الأب، يمكنك استدعاء دوال الأبناء المختلفة في وقت التشغيل. هذا هو "dynamic dispatch".

C++ — Polymorphism نظام الأشكالshapes.cpp 
#include <vector>
#include <memory>

class Shape {
protected:
    string color;
    string name;

public:
    Shape(string c, string n) : color(c), name(n) {}

    virtual double area()      const = 0;  // Pure virtual
    virtual double perimeter() const = 0;  // Pure virtual
    virtual void   draw()      const {
        cout << "🎨 رسم " << name << " باللون " << color << "\n";
    }

    void printInfo() const {
        cout << "───────────────────\n";
        cout << name << " (" << color << ")\n";
        cout << "المساحة: "   << area()      << " وحدة²\n";
        cout << "المحيط: "   << perimeter() << " وحدة\n";
    }

    virtual ~Shape() = default;
};

// ── الأبناء ────────────────────────────────────────
class Circle : public Shape {
    double radius;
public:
    Circle(string c, double r) : Shape(c, "دائرة"), radius(r) {}
    double area()      const override { return M_PI * radius * radius; }
    double perimeter() const override { return 2 * M_PI * radius; }
    void draw() const override {
        Shape::draw();
        cout << "  ⭕ نصف القطر: " << radius << "\n";
    }
};

class Rectangle : public Shape {
    double w, h;
public:
    Rectangle(string c, double w, double h)
        : Shape(c, "مستطيل"), w(w), h(h) {}
    double area()      const override { return w * h; }
    double perimeter() const override { return 2 * (w + h); }
};

class Triangle : public Shape {
    double a, b, c;
public:
    Triangle(string col, double a, double b, double c)
        : Shape(col, "مثلث"), a(a), b(b), c(c) {}
    double perimeter() const override { return a+b+c; }
    double area() const override {
        double s = perimeter() / 2;   // Heron's formula
        return sqrt(s*(s-a)*(s-b)*(s-c));
    }
};

// ── الجوهر: Polymorphism ────────────────────────────
int main() {
    // مجموعة من أنواع مختلفة — Polymorphic collection!
    vector<unique_ptr<Shape>> shapes;
    shapes.push_back(make_unique<Circle>("أحمر", 5));
    shapes.push_back(make_unique<Rectangle>("أزرق", 4, 6));
    shapes.push_back(make_unique<Triangle>("أخضر", 3, 4, 5));

    // نفس الاستدعاء — سلوكيات مختلفة تلقائياً!
    double totalArea = 0;
    for (const auto& s : shapes) {
        s->printInfo();
        s->draw();
        totalArea += s->area();
    }

    cout << "إجمالي المساحة: " << totalArea << " وحدة²\n";
}

Polymorphic Collection — نخزن أنواعاً مختلفة في مصفوفة واحدة بـ pointer للأب
virtual Destructor — ضروري دائماً مع الوراثة لضمان تحرير الذاكرة الصحيح

C++ — Abstract Class & Interfacedatabase.cpp 
// ── Interface: عقد خالص ────────────────────────────
class IDatabase {
public:
    virtual bool   connect(const string& connStr) = 0;
    virtual bool   disconnect()                    = 0;
    virtual bool   execute(const string& query)   = 0;
    virtual string getType() const                 = 0;
    virtual ~IDatabase() = default;
};

// ── Abstract Base: يضيف سلوكاً مشتركاً ────────────
class DatabaseBase : public IDatabase {
protected:
    bool   isConnected = false;
    int    queryCount  = 0;
    string host;

public:
    void logQuery(const string& q) {
        queryCount++;
        cout << "📊 [Query #" << queryCount << "] " << q << "\n";
    }
    int getQueryCount() const { return queryCount; }
    // execute لا يزال abstract — الأبناء ينفذونه
};

// ── Concrete: MySQL ─────────────────────────────────
class MySQLDatabase : public DatabaseBase {
public:
    bool connect(const string& connStr) override {
        host = connStr;
        isConnected = true;
        cout << "🐬 MySQL: متصل بـ " << connStr << "\n";
        return true;
    }
    bool disconnect() override {
        isConnected = false;
        cout << "🐬 MySQL: انقطع الاتصال\n";
        return true;
    }
    bool execute(const string& q) override {
        if (!isConnected) { cout << "❌ غير متصل!\n"; return false; }
        logQuery(q);
        return true;
    }
    string getType() const override { return "MySQL"; }
};

// ── الاستخدام ───────────────────────────────────────
void runQueries(IDatabase& db) {
    // هذه الدالة تعمل مع أي قاعدة بيانات!
    db.execute("SELECT * FROM users");
    db.execute("INSERT INTO orders...");
    db.execute("UPDATE inventory...");
}

int main() {
    MySQLDatabase mysql;
    mysql.connect("localhost:3306");
    runQueries(mysql);
    mysql.disconnect();
}

9.1 Function Template

C++ — Function Templatestemplates.cpp 
// ✅ دالة واحدة تعمل مع أي نوع
template<typename T>
T maximum(T a, T b) {
    return (a > b) ? a : b;
}

// Template مع نوعين مختلفين
template<typename T, typename U>
void printPair(T first, U second) {
    cout << "(" << first << ", " << second << ")\n";
}

// Template مع قيمة افتراضية
template<typename T, int SIZE = 10>
void printArray(T arr[]) {
    for (int i = 0; i < SIZE; i++)
        cout << arr[i] << " ";
    cout << "\n";
}

int main() {
    cout << maximum(3, 7)         << "\n"; // 7 (int)
    cout << maximum(3.14, 2.71)  << "\n"; // 3.14 (double)
    cout << maximum('Z', 'A')     << "\n"; // Z (char)
    printPair("أحمد", 25);                // (أحمد, 25)
}

9.2 Class Template — Stack جنريك

C++ — Generic Stackgeneric_stack.cpp 
template<typename T, int MAX_SIZE = 100>
class Stack {
private:
    T   data[MAX_SIZE];
    int top;

public:
    Stack() : top(-1) {}

    bool push(const T& item) {
        if (isFull()) { cout << "❌ Stack ممتلئ!\n"; return false; }
        data[++top] = item;
        return true;
    }

    T pop() {
        if (isEmpty()) throw runtime_error("Stack فارغ!");
        return data[top--];
    }

    T&   peek()    const { return data[top]; }
    bool isEmpty() const { return top == -1; }
    bool isFull()  const { return top == MAX_SIZE - 1; }
    int  size()    const { return top + 1; }

    void print() const {
        cout << "Stack [";
        for (int i = top; i >= 0; i--)
            cout << data[i] << (i ? ", " : "");
        cout << "] (top→bottom)\n";
    }
};

int main() {
    Stack<int>    intStack;
    Stack<string> strStack;
    Stack<double, 5> smallStack;

    intStack.push(10); intStack.push(20); intStack.push(30);
    intStack.print();   // Stack [30, 20, 10]

    strStack.push("مرحبا");
    strStack.push("عالم");
    strStack.print();   // Stack [عالم, مرحبا]
}

#include <iostream>
#include <vector>

// ══════════════════════════════════════════════
   Abstract Base
   ══════════════════════════════════════════════
class Person {
protected:
    string name, phone, nationalId;
    int    age;

public:
    Person(string n, string ph, string id, int a)
        : name(n), phone(ph), nationalId(id), age(a) {}

    virtual string getRole()   const = 0;
    virtual void   printInfo() const {
        cout << "[" << getRole() << "] "
             << name  << " | "
             << age   << " سنة | "
             << phone << "\n";
    }

    string getName() const { return name; }
    virtual ~Person() = default;
};

// ══════════════════════════════════════════════
   Staff Hierarchy
   ══════════════════════════════════════════════
class Staff : public Person {
protected:
    double baseSalary;
    string employeeId;

public:
    Staff(string n, string ph, string nid, int a, double sal, string eid)
        : Person(n, ph, nid, a), baseSalary(sal), employeeId(eid) {}

    virtual double getSalary() const { return baseSalary; }
};

class Doctor : public Staff {
private:
    string specialization;
    int    yearsExperience;
    double consultationFee;

public:
    Doctor(string n, string ph, string nid, int a,
            double sal, string eid, string spec, int exp, double fee)
        : Staff(n, ph, nid, a, sal, eid),
          specialization(spec), yearsExperience(exp), consultationFee(fee) {}

    string getRole() const override { return "طبيب"; }

    double getSalary() const override {
        return baseSalary + (yearsExperience * 500);
    }

    void printInfo() const override {
        Person::printInfo();
        cout << "  التخصص: " << specialization
             << " | الخبرة: " << yearsExperience << " سنة\n";
        cout << "  رسم الكشف: " << consultationFee << " جنيه\n";
    }

    string getSpecialization() const { return specialization; }
    double getConsultationFee() const { return consultationFee; }
};

// ══════════════════════════════════════════════
   Patient
   ══════════════════════════════════════════════
class Patient : public Person {
private:
    string patientId;
    string bloodType;
    vector<string> allergies;
    vector<string> medicalHistory;
    double totalBill;

public:
    Patient(string n, string ph, string nid, int a,
             string pid, string blood)
        : Person(n, ph, nid, a),
          patientId(pid), bloodType(blood), totalBill(0) {}

    string getRole() const override { return "مريض"; }

    void addAllergy(string a)      { allergies.push_back(a); }
    void addHistory(string h)      { medicalHistory.push_back(h); }
    void addToBill(double amount)  { totalBill += amount; }

    void printInfo() const override {
        Person::printInfo();
        cout << "  فصيلة الدم: " << bloodType
             << " | الفاتورة: " << totalBill << " جنيه\n";
        if (!allergies.empty()) {
            cout << "  ⚠️ حساسية من: ";
            for (auto& a : allergies) cout << a << " ";
            cout << "\n";
        }
    }

    string getPatientId() const { return patientId; }
    double getTotalBill() const { return totalBill; }
};

class Appointment {
private:
    string   appointmentId, date, time, notes;
    Doctor*  doctor;
    Patient* patient;
    bool     completed;

public:
    Appointment(string id, Doctor* d, Patient* p,
                 string dt, string tm)
        : appointmentId(id), doctor(d), patient(p),
          date(dt), time(tm), completed(false) {}

    void complete(string diagnosis) {
        completed = true;
        notes = diagnosis;
        // تضاف رسوم الكشف للفاتورة
        patient->addToBill(doctor->getConsultationFee());
        patient->addHistory(diagnosis);
        cout << "✅ انتهى الموعد: " << diagnosis << "\n";
    }

    void print() const {
        cout << "📅 موعد [" << appointmentId << "] "
             << date << " " << time << "\n";
        cout << "   الطبيب: "  << doctor->getName() << "\n";
        cout << "   المريض: " << patient->getName() << "\n";
        cout << "   الحالة: "  << (completed ? "✅ مكتمل" : "⏳ قيد الانتظار") << "\n";
    }
};

class Hospital {
private:
    string name;
    vector<unique_ptr<Doctor>>      doctors;
    vector<unique_ptr<Patient>>     patients;
    vector<unique_ptr<Appointment>> appointments;
    int nextAppId = 1000;

public:
    Hospital(string n) : name(n) {}

    Doctor* addDoctor(string n, string ph, string nid, int age,
                        double sal, string eid, string spec, int exp, double fee) {
        doctors.push_back(make_unique<Doctor>(n, ph, nid, age, sal, eid, spec, exp, fee));
        cout << "✅ تم تسجيل الطبيب: " << n << "\n";
        return doctors.back().get();
    }

    Patient* addPatient(string n, string ph, string nid,
                          int age, string pid, string blood) {
        patients.push_back(make_unique<Patient>(n, ph, nid, age, pid, blood));
        cout << "✅ تم تسجيل المريض: " << n << "\n";
        return patients.back().get();
    }

    Appointment* bookAppointment(Doctor* d, Patient* p,
                                     string date, string time) {
        string id = "APP" + to_string(nextAppId++);
        appointments.push_back(make_unique<Appointment>(id, d, p, date, time));
        cout << "📅 موعد محجوز: " << id << "\n";
        return appointments.back().get();
    }

    void printDoctors() const {
        cout << "═══ أطباء " << name << " ═══\n";
        for (const auto& d : doctors) d->printInfo();
    }

    void printPatients() const {
        cout << "═══ مرضى " << name << " ═══\n";
        for (const auto& p : patients) p->printInfo();
    }

    double getTotalRevenue() const {
        double total = 0;
        for (const auto& p : patients)
            total += p->getTotalBill();
        return total;
    }
};
