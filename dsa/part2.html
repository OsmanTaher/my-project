10.3 تمارين LeetCode Style

مسألة 1: Kth Largest Element in ArrayMedium AmazonGoogleMeta

المشكلة: أوجد العنصر الـ k الأكبر في المصفوفة.

مثال: [3,2,1,5,6,4], k=2 → 5

💡استخدم Min-Heap بحجم k. إذا تجاوز الحجم k، احذف أصغر عنصر. الناتج: top() هو الـ k الأكبر.
C++ — Min-Heap O(n log k) 
int findKthLargest(vector<int>& nums, int k) {
    priority_queue<int, vector<int>, greater<int>> minH;

    for (int n : nums) {
        minH.push(n);
        if (minH.size() > k) minH.pop(); // احذف الأصغر
    }
    return minH.top(); // الـ k الأكبر
}
// Time: O(n log k) | Space: O(k)
مسألة 2: Top K Frequent ElementsMedium AmazonGoogleMetaMicrosoft

المشكلة: أوجد الـ k عناصر الأكثر تكراراً.

مثال: [1,1,1,2,2,3], k=2 → [1,2]

C++ — HashMap + Min-Heap O(n log k) 
vector<int> topKFrequent(vector<int>& nums, int k) {
    unordered_map<int,int> freq;
    for (int n : nums) freq[n]++;

    // Min-Heap: {frequency, value}
    priority_queue<pair<int,int>,
        vector<pair<int,int>>,
        greater<>> minH;

    for (auto& [val, cnt] : freq) {
        minH.push({cnt, val});
        if (minH.size() > k) minH.pop();
    }

    vector<int> res;
    while (!minH.empty()) {
        res.push_back(minH.top().second);
        minH.pop();
    }
    return res;
}

مسألة 1: Longest Consecutive SequenceMedium GoogleAmazonMeta

المشكلة: [100,4,200,1,3,2] → أوجد طول أطول تسلسل متتالي. الجواب: 4 ([1,2,3,4])

💡ضع كل الأرقام في Set. ابحث فقط عن بداية التسلسل (العدد الذي n-1 غير موجود في الـ Set).
C++ — O(n) باستخدام HashSet 
int longestConsecutive(vector<int>& nums) {
    unordered_set<int> s(nums.begin(), nums.end());
    int longest = 0;

    for (int n : s) {
        if (!s.count(n - 1)) {    // بداية تسلسل!
            int cur = n, len = 1;
            while (s.count(cur + 1)) { cur++; len++; }
            longest = max(longest, len);
        }
    }
    return longest;
}
// Time: O(n) | Space: O(n)

vector<vector<string>> groupAnagrams(vector<string>& strs) {
    unordered_map<string, vector<string>> groups;

    for (string& s : strs) {
        string key = s;
        sort(key.begin(), key.end()); // المفتاح = الكلمة مرتبة
        groups[key].push_back(s);
    }

    vector<vector<string>> res;
    for (auto& [_, group] : groups)
        res.push_back(group);
    return res;
}

13.1 ركنا Recursion

1. Base Case ⚓

الحالة التي تُوقف الاستدعاء. بدونها → Stack Overflow!

2. Recursive Case 🔄

استدعاء الدالة بمشكلة أصغر تقترب من الـ Base Case.

3. Trust the Magic ✨

افترض أن الاستدعاء يعمل صح وابنِ عليه.

C++ — Factorial & Fibonacci 
// ── Factorial: n! ──────────────────────────────
int factorial(int n) {
    if (n <= 1) return 1;         // Base Case
    return n * factorial(n - 1);  // Recursive
}
// factorial(5) → 5*4*3*2*1 = 120

// ── Fibonacci Memoization O(n) ─────────────────
unordered_map<int,int> memo;
int fib(int n) {
    if (n <= 1) return n;
    if (memo.count(n)) return memo[n];
    return memo[n] = fib(n-1) + fib(n-2);
}
// بدون مemoization: O(2^n) | معها: O(n)

مسألة 1: Subsets (Power Set)Medium AmazonMicrosoftGoogle

المشكلة: [1,2,3] → أرجع كل المجموعات الجزئية الممكنة (8 مجموعات = 2³)

C++ — Backtracking O(2^n) 
void bt(vector<int>& nums, int start,
         vector<int>& cur, vector<vector<int>>& res) {
    res.push_back(cur);

    for (int i = start; i < nums.size(); i++) {
        cur.push_back(nums[i]);  // اختر
        bt(nums, i+1, cur, res); // استكشف
        cur.pop_back();          // تراجع!
    }
}

vector<vector<int>> subsets(vector<int>& nums) {
    vector<vector<int>> res;
    vector<int> cur;
    bt(nums, 0, cur, res);
    return res;
}
مسألة 2: PermutationsMedium MetaAmazonGoogle

المشكلة: [1,2,3] → أرجع كل التبديلات الممكنة (6 = 3!)

C++ — Backtracking O(n!) 
void permute(vector<int>& nums, vector<bool>& used,
              vector<int>& cur, vector<vector<int>>& res) {
    if (cur.size() == nums.size()) {
        res.push_back(cur); return;
    }
    for (int i = 0; i < nums.size(); i++) {
        if (used[i]) continue;
        used[i] = true;
        cur.push_back(nums[i]);
        permute(nums, used, cur, res);
        cur.pop_back();
        used[i] = false; // تراجع!
    }
}

C++ — Bottom-Up DP O(amount × coins) 
int coinChange(vector<int>& coins, int amount) {
    vector<int> dp(amount+1, amount+1); // ∞
    dp[0] = 0;

    for (int i = 1; i <= amount; i++)
        for (int coin : coins)
            if (coin <= i)
                dp[i] = min(dp[i], 1 + dp[i-coin]);

    return dp[amount] > amount ? -1 : dp[amount];
}
// Time: O(amount × coins.size()) | Space: O(amount)
مسألة 2: Longest Common Subsequence (LCS)Medium GoogleAmazonMicrosoft

المشكلة: text1="abcde", text2="ace" → أطول تسلسل مشترك = 3 ("ace")

C++ — 2D DP Table O(m×n) 
int lcs(string t1, string t2) {
    int m = t1.size(), n = t2.size();
    vector<vector<int>> dp(m+1, vector<int>(n+1, 0));

    for (int i = 1; i <= m; i++)
        for (int j = 1; j <= n; j++) {
            if (t1[i-1] == t2[j-1])
                dp[i][j] = dp[i-1][j-1] + 1;
            else
                dp[i][j] = max(dp[i-1][j], dp[i][j-1]);
        }
    return dp[m][n];
}
مسألة 3: House RobberMedium AmazonGoogleMeta

int rob(vector<int>& nums) {
    int prev2 = 0, prev1 = 0;

    for (int n : nums) {
        int cur = max(prev1, prev2 + n);
        prev2 = prev1;
        prev1 = cur;
    }
    return prev1;
}
// dp[i] = max(dp[i-1], dp[i-2] + nums[i])

bool canJump(vector<int>& nums) {

void bfs(vector<vector<int>>& g, int start) {
    vector<bool> vis(g.size(), false);
    queue<int> q;
    vis[start] = true;
    q.push(start);

    while (!q.empty()) {
        int node = q.front(); q.pop();
        cout << node << " ";
        for (int nb : g[node])
            if (!vis[nb]) {
                vis[nb] = true;
                q.push(nb);
            }
    }
}
// Time: O(V+E) | Space: O(V)

void dfs(vector<vector<int>>& g,

    vis[node] = true;

16.3 Dijkstra — أقصر مسار ⭐⭐

يجد أقصر مسار من عقدة مصدر لكل العقد في الـ Weighted Graph (بدون حواف سالبة).

C++ — Dijkstra O((V+E) log V)dijkstra.cpp 
vector<int> dijkstra(vector<vector<pair<int,int>>>& g, int src) {
    int V = g.size();
    vector<int> dist(V, INT_MAX);
    // Min-Heap: {distance, node}
    priority_queue<pair<int,int>,
        vector<pair<int,int>>, greater<>> pq;

    dist[src] = 0;
    pq.push({0, src});

    while (!pq.empty()) {
        auto [d, u] = pq.top(); pq.pop();
        if (d > dist[u]) continue; // تجاهل قديم

        for (auto [v, w] : g[u]) {
            if (dist[u] + w < dist[v]) {
                dist[v] = dist[u] + w;
                pq.push({dist[v], v});
            }
        }
    }
    return dist; // المسافة لكل عقدة
}

void sink(vector<vector<char>>& g, int r, int c) {
    if (r<0||r>=g.size()||c<0||c>=g[0].size()||g[r][c]!='1')
        return;
    g[r][c] = '0'; // غرّق الأرض
    sink(g,r+1,c); sink(g,r-1,c);
    sink(g,r,c+1); sink(g,r,c-1);
}

int numIslands(vector<vector<char>>& grid) {
    int islands = 0;
    for (int r = 0; r < grid.size(); r++)
        for (int c = 0; c < grid[0].size(); c++)
            if (grid[r][c]=='1') { sink(grid,r,c); islands++; }
    return islands;
}
