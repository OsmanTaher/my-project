CHAPTER 01 — HARDWARE OVERVIEW
مكونات الحاسوب
Hardware Components
قبل أن تكتب سطراً واحداً من الكود — لا بدّ أن تفهم الآلة التي ستنفّذه. هذا الفصل يشرح كل مكوّن من مكونات الحاسوب ودوره الحقيقي.

1.1 نظرة عامة على الحاسوب

💡 تعريف الحاسوب

الحاسوب هو جهاز إلكتروني يقبل البيانات (Input)، ويعالجها (Processing)، ويُخرج النتائج (Output)، ويمكنه تخزين البيانات (Storage). كل هذا يحدث وفق تعليمات مبرمجة مسبقاً.

VON NEUMANN ARCHITECTURE — معمارية فون نيومان

1.2 المكونات الرئيسية بالتفصيل

وحدة المعالجة المركزية — CPU

المعالج هو "عقل" الحاسوب. يُنفّذ التعليمات الواحدة تلو الأخرى بسرعة هائلة. يُقاس أداؤه بـ الجيجاهرتز (GHz) — عدد دورات العمل في الثانية.

وحدة الحساب والمنطق (ALU)

تُنفّذ العمليات الحسابية (+، -، ×، ÷) والمنطقية (AND, OR, NOT). كل ما يحدث في البرنامج يمر من هنا.

وحدة التحكم (CU)

تقرأ التعليمات من الذاكرة وتفسّرها وتُوجّه بقية المكونات لتنفيذها. هي "المايسترو" الذي ينسّق الكل.

التسجيلات (Registers)

ذاكرة صغيرة جداً داخل المعالج نفسه — أسرع نوع من الذاكرة. تخزّن البيانات التي يعمل عليها المعالج حالياً.

الذاكرة العشوائية — RAM

ذاكرة مؤقتة تُستخدم أثناء تشغيل البرامج. سريعة، لكنها تُفقد محتواها عند إيقاف التيار. كلما زادت سعتها، استطاع الحاسوب تشغيل برامج أكثر في آنٍ واحد.

📌مثال عملي: عندما تفتح ملف Word، يُنسخ الملف من القرص الصلب (تخزين دائم) إلى RAM (ذاكرة مؤقتة) ليتمكن المعالج من الوصول إليه بسرعة.

التخزين الدائم — Storage

القرص الصلب HDD

أقراص مغناطيسية دوّارة. سعة كبيرة، سعر منخفض، لكنه أبطأ من SSD.

محرك الحالة الصلبة SSD

ذاكرة فلاش — لا أجزاء متحركة. أسرع بكثير من HDD، أكثر متانة، سعر أعلى.

CHAPTER 04 — NUMBER SYSTEMS & BINARY
أنظمة الأعداد
Number Systems & Binary Arithmetic
الحاسوب يفهم لغة واحدة فقط: الصفر والواحد. هذا الفصل يشرح كيف نُحوّل أي عدد إلى ثنائي، وكيف يخزّن الحاسوب البيانات في ذاكرته.

4.1 أنظمة الأعداد الأربعة

النظام الأساس الأرقام المستخدمة مثال
عشري Decimal 10 0-9 255₁₀
ثنائي Binary 2 0, 1 11111111₂
ثماني Octal 8 0-7 377₈
سداسي عشر Hex 16 0-9, A-F FF₁₆
لماذا ثنائي؟

الترانزستور (مفتاح إلكتروني) له حالتان فقط:
✦ 1 = تيار موجود (مفتوح)
✦ 0 = لا تيار (مغلق)

ملايين الترانزستورات × حالتان = بيانات لا نهاية لها!

4.2 التحويل بين الأنظمة

التحويلات الأساسية — مع أمثلة 
══ تحويل عشري → ثنائي (بالقسمة على 2) ══
255 ÷ 2 = 127  باقي  1
127 ÷ 2 = 63   باقي  1
63  ÷ 2 = 31   باقي  1
31  ÷ 2 = 15   باقي  1
15  ÷ 2 = 7    باقي  1
7   ÷ 2 = 3    باقي  1
3   ÷ 2 = 1    باقي  1
1   ÷ 2 = 0    باقي  1
↑ اقرأ البواقي من أسفل لأعلى: 11111111
∴ 255₁₀ = 11111111₂

══ تحويل ثنائي → عشري (الأوزان) ══
1011₂ = 1×2³ + 0×2² + 1×2¹ + 1×2⁰
       = 8   +  0   +  2   +  1
       = 11₁₀

══ تحويل عشري → سداسي عشر ══
255 ÷ 16 = 15  باقي  15 → F
15  ÷ 16 = 0   باقي  15 → F
∴ 255₁₀ = FF₁₆  (لذلك الأبيض في CSS هو #FFFFFF)

══ ثنائي → سداسي عشر (كل 4 بتات = رقم واحد) ══
1111 1111₂  →  F    F   ₁₆  =  FF₁₆
1010 0101₂  →  A    5   ₁₆  =  A5₁₆

10.1 ما هو المؤشر؟

💡 التعريف

المؤشر (Pointer) هو متغير يحمل عنوان متغير آخر في الذاكرة. بدلاً من تخزين قيمة، يُخزّن عنواناً (رقم البايت في الـ RAM).

POINTER CONCEPT — تصوّر المؤشر

10.2 عمليات المؤشر

C++ — Pointer Operationspointers.cpp 
#include <iostream>
using namespace std;

int main() {
    int age = 25;         // متغير عادي
    int* ptr = &age;      // مؤشر يحمل عنوان age
    //        ↑ & = عامل "عنوان" (Address-of operator)

    cout << "قيمة age     : " << age   << endl;  // 25
    cout << "عنوان age    : " << &age  << endl;  // 0x...  (العنوان)
    cout << "قيمة ptr     : " << ptr   << endl;  // 0x...  (نفس العنوان)
    cout << "القيمة بالمؤشر: " << *ptr  << endl;  // 25
    //                                ↑ * = عامل "إلغاء الإشارة" (Dereference)

    // تعديل القيمة عبر المؤشر
    *ptr = 30;
    cout << "age بعد *ptr=30: " << age << endl;  // 30 — تغيّرت!

    // حساب المؤشر (Pointer Arithmetic)
    int arr[] = {10, 20, 30, 40, 50};
    int* p = arr;              // يشير للعنصر الأول

    cout << "arr[0] = " << *p     << endl;  // 10
    cout << "arr[1] = " << *(p+1) << endl;  // 20
    cout << "arr[2] = " << *(p+2) << endl;  // 30
    // p+1 يتقدم بـ sizeof(int) = 4 بايت في الذاكرة

    return 0;
}

C++ — Swap using Pointers 
void swap(int* a, int* b) {
    int temp = *a;   // احفظ قيمة a
    *a = *b;          // ضع قيمة b في a
    *b = temp;        // ضع القيمة المحفوظة في b
}

int main() {
    int x = 10, y = 20;
    cout << "قبل: x=" << x << " y=" << y << endl; // x=10 y=20

    swap(&x, &y);   // نمرر العناوين لا القيم

    cout << "بعد: x=" << x << " y=" << y << endl; // x=20 y=10
    return 0;
}

C++ — Inventory Systeminventory.cpp 
#include <iostream>
#include <string>
using namespace std;

struct Product {
    string name;
    double price;
    int    quantity;
    string barcode;
};

int productCount = 0;
Product store[10];

void addProduct(string n, double p, int q, string b) {
    if (productCount < 10) {
        store[productCount] = {n, p, q, b};
        productCount++;
        cout << "✅ أُضيف: " << n << endl;
    }
}

bool sellProduct(string barcode, int qty) {
    for (int i = 0; i < productCount; i++) {
        if (store[i].barcode == barcode) {
            if (store[i].quantity >= qty) {
                store[i].quantity -= qty;
                cout << "✅ بيع " << qty << " من " << store[i].name << endl;
                return true;
            }
            cout << "❌ الكمية غير كافية!\n";
            return false;
        }
    }
    cout << "❌ الباركود غير موجود!\n";
    return false;
}

void lowStockReport(int threshold) {
    cout << "═══ منتجات كمّيتها أقل من " << threshold << " ═══\n";
    for (int i = 0; i < productCount; i++) {
        if (store[i].quantity < threshold) {
            cout << "⚠️ " << store[i].name
                 << " — كمية: " << store[i].quantity << endl;
        }
    }
}

int main() {
